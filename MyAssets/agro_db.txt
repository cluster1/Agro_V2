/* ---------------------------
   DROP EXISTING TABLES (opsional)
   Urutan drop dibuat agar tidak terjadi dependency error
   --------------------------- */
drop table if exists tm_livestock_activity;
drop table if exists tm_livestock;
drop table if exists tm_site_media;
drop table if exists tm_weather;
drop table if exists tm_sensor_read_update;
drop table if exists tm_sensor_read;
drop table if exists td_device_sensor;
drop table if exists td_sensor;
drop table if exists tm_device;
drop table if exists tm_cluster;
drop table if exists tm_site;
drop table if exists tm_user;
drop table if exists tm_tenant;
drop table if exists tm_log_receive;
drop table if exists tm_alarm;
drop table if exists tr_alarm_code;
drop table if exists tr_kode_telemetri;
drop table if exists tr_unit;

/* ===============================================================
   TABEL: tm_tenant - Tenant / Organisasi
   =============================================================== */
create table tm_tenant
(
   tenant_id            varchar(32) not null,
   tenant_name          varchar(128),
   tenant_type          varchar(32),         -- koperasi / perusahaan / petani_mandiri
   tenant_address       varchar(255),
   tenant_contact       varchar(64),
   tenant_logo          varchar(255),
   tenant_created       datetime,
   tenant_update        datetime,
   primary key (tenant_id)
);

/* ===============================================================
   TABEL: tm_user - Pengguna multi-role dan multi-tenant
   =============================================================== */
create table tm_user
(
   user_id              varchar(32) not null,
   tenant_id            varchar(32),
   user_name            varchar(64),
   user_pass            varchar(128),        -- catatan: simpan hash, bukan plain text
   user_email           varchar(128),
   user_phone           varchar(64),
   user_role            varchar(32),         -- admin / farmer / investor / viewer
   user_level           int,                 -- 0 = global admin, 1 = tenant admin, 2 = user biasa
   user_sts             varchar(32),
   user_created         datetime,
   user_update          datetime,
   primary key (user_id),
   foreign key (tenant_id) references tm_tenant(tenant_id)
);

/* ===============================================================
   TABEL: tm_site - Lokasi utama (Farm / Field / Ranch)
   =============================================================== */
create table tm_site
(
   site_id              varchar(32) not null,
   tenant_id            varchar(32),
   site_name            varchar(64),
   site_type            varchar(16),         -- FARM / FIELD / RANCH / GREENHOUSE
   site_address         varchar(128),
   site_lon             double,
   site_lat             double,
   site_alt             double,
   site_sts             int(1),
   site_created         datetime,
   site_update          datetime,
   primary key (site_id),
   foreign key (tenant_id) references tm_tenant(tenant_id)
);

/* ===============================================================
   TABEL: tm_cluster - Unit operasional (kandang, kolam, blok lahan)
   =============================================================== */
create table tm_cluster
(
   cluster_id           varchar(32) not null,
   tenant_id            varchar(32),
   site_id              varchar(32),
   cluster_name         varchar(64),
   cluster_type         varchar(32),         -- HOUSE / POND / FIELD / GREENHOUSE
   cluster_capacity     int,
   cluster_current      int,
   cluster_desc         varchar(255),
   cluster_created      datetime,
   cluster_update       datetime,
   primary key (cluster_id),
   foreign key (tenant_id) references tm_tenant(tenant_id),
   foreign key (site_id) references tm_site(site_id)
);

/* ===============================================================
   TABEL: tm_device - Perangkat IoT (node / controller)
   =============================================================== */
create table tm_device
(
   dev_id               varchar(32) not null,
   tenant_id            varchar(32),
   site_id              varchar(32),
   cluster_id           varchar(32),
   user_id              varchar(32),
   dev_name             varchar(64),
   dev_token            varchar(128),
   dev_location         varchar(128),
   dev_lon              double,
   dev_lat              double,
   dev_alt              double,
   dev_number_id        varchar(64),
   dev_ip               varchar(32),
   dev_port             varchar(64),
   dev_img              varchar(128),
   dev_sts              int(1),
   dev_created          datetime,
   dev_update           datetime,
   primary key (dev_id),
   foreign key (tenant_id) references tm_tenant(tenant_id),
   foreign key (site_id) references tm_site(site_id),
   foreign key (cluster_id) references tm_cluster(cluster_id),
   foreign key (user_id) references tm_user(user_id)
);

/* ===============================================================
   TABEL: td_sensor - Master sensor
   =============================================================== */
create table td_sensor
(
   sens_id              varchar(32) not null,
   tenant_id            varchar(32),
   dev_id               varchar(32),
   sens_name            varchar(64),
   sens_address         varchar(32),
   sens_location        varchar(64),
   sens_lat             double,
   sens_lon             double,
   sens_alt             double,
   primary key (sens_id),
   foreign key (tenant_id) references tm_tenant(tenant_id),
   foreign key (dev_id) references tm_device(dev_id)
);

/* ===============================================================
   TABEL: td_device_sensor - Kaitan device dan sensor, threshold dsb
   =============================================================== */
create table td_device_sensor
(
   ds_id                varchar(32) not null,
   dev_id               varchar(32) not null,
   tenant_id            varchar(32),
   unit_id              varchar(32),
   sens_id              varchar(32),
   cluster_id           varchar(32),
   animal_id            varchar(32),         -- opsional: sensor menempel pada ternak
   dc_normal_value      float,
   ds_min_norm_value    float,
   ds_max_norm_value    float,
   ds_min_value         float,
   ds_max_value         float,
   ds_min_val_warn      float,
   ds_max_val_warn      float,
   ds_name              varchar(128),
   ds_address           varchar(32),
   ds_seq               int(4),
   ds_sts               int(1),
   ds_created           datetime,
   ds_update            datetime,
   primary key (ds_id, dev_id),
   foreign key (tenant_id) references tm_tenant(tenant_id),
   foreign key (dev_id) references tm_device(dev_id),
   foreign key (sens_id) references td_sensor(sens_id),
   foreign key (cluster_id) references tm_cluster(cluster_id)
);

/* ===============================================================
   TABEL: tm_sensor_read - Data bacaan sensor (time-series)
   =============================================================== */
create table tm_sensor_read
(
   read_id              varchar(64) not null,
   ds_id                varchar(32),
   dev_id               varchar(32),
   log_rx_id            varchar(32),
   tenant_id            varchar(32),
   cluster_id           varchar(32),
   read_date            datetime,
   read_value           varchar(128),
   read_sts             int(1),
   primary key (read_id),
   foreign key (tenant_id) references tm_tenant(tenant_id),
   foreign key (cluster_id) references tm_cluster(cluster_id)
);

/* ===============================================================
   TABEL: tm_sensor_read_update - Nilai update terakhir (cache)
   =============================================================== */
create table tm_sensor_read_update
(
   read_update_id       varchar(64) not null,
   ds_id                varchar(32),
   dev_id               varchar(32),
   tenant_id            varchar(32),
   cluster_id           varchar(32),
   read_update_date     datetime,
   read_update_value    varchar(128),
   primary key (read_update_id),
   foreign key (tenant_id) references tm_tenant(tenant_id),
   foreign key (cluster_id) references tm_cluster(cluster_id)
);

/* ===============================================================
   TABEL: tr_alarm_code - Kode jenis alarm
   =============================================================== */
create table tr_alarm_code
(
   alcode_id            varchar(32) not null,
   alcode_note          varchar(128),
   alcode_sts           int(1),
   alcode_update        datetime,
   primary key (alcode_id)
);

/* ===============================================================
   TABEL: tm_alarm - Alarm kondisi abnormal
   =============================================================== */
create table tm_alarm
(
   alm_id               varchar(32) not null,
   alcode_id            varchar(32),
   read_id              varchar(64),
   tenant_id            varchar(32),
   cluster_id           varchar(32),
   alm_date             datetime,
   primary key (alm_id),
   foreign key (alcode_id) references tr_alarm_code(alcode_id),
   foreign key (tenant_id) references tm_tenant(tenant_id),
   foreign key (cluster_id) references tm_cluster(cluster_id)
);

/* ===============================================================
   TABEL: tm_log_receive - Log payload MQTT / HTTP
   =============================================================== */
create table tm_log_receive
(
   log_rx_id            varchar(32) not null,
   tenant_id            varchar(32),
   log_rx_date          datetime,
   log_rx_param_id      varchar(32),
   log_rx_payload       varchar(1024),
   primary key (log_rx_id),
   foreign key (tenant_id) references tm_tenant(tenant_id)
);

/* ===============================================================
   TABEL: tr_kode_telemetri - Kamus parameter telemetri
   =============================================================== */
create table tr_kode_telemetri
(
   kt_id                varchar(64) not null,
   kt_name              varchar(64),
   kt_norm_value        varchar(32),
   kt_min_value         varchar(32),
   kt_max_value         varchar(32),
   kt_min_value_warn    varchar(32),
   kt_max_value_warn    varchar(32),
   kt_sts               int(1),
   primary key (kt_id)
);

/* ===============================================================
   TABEL: tr_unit - Satuan
   =============================================================== */
create table tr_unit
(
   unit_id              varchar(32) not null,
   unit_name            varchar(64),
   unit_symbol          varchar(8),
   unit_sts             int(1),
   unit_update          datetime,
   primary key (unit_id)
);

/* ===============================================================
   TABEL: tm_weather - Data cuaca eksternal (BMKG atau source lain)
   =============================================================== */
create table tm_weather
(
   weather_id           varchar(64) not null,
   tenant_id            varchar(32),
   site_id              varchar(32),
   weather_date         datetime,
   temperature          float,
   humidity             float,
   rainfall             float,
   wind_speed           float,
   weather_source       varchar(64),
   primary key (weather_id),
   foreign key (tenant_id) references tm_tenant(tenant_id),
   foreign key (site_id) references tm_site(site_id)
);

/* ===============================================================
   TABEL: tm_site_media - Foto/Video kondisi lahan/kandang
   =============================================================== */
create table tm_site_media
(
   media_id             varchar(64) not null,
   tenant_id            varchar(32),
   site_id              varchar(32),
   cluster_id           varchar(32),
   user_id              varchar(32),
   media_type           varchar(16),  -- photo / video
   media_path           varchar(255),
   media_desc           varchar(255),
   media_date           datetime,
   primary key (media_id),
   foreign key (tenant_id) references tm_tenant(tenant_id),
   foreign key (site_id) references tm_site(site_id),
   foreign key (cluster_id) references tm_cluster(cluster_id),
   foreign key (user_id) references tm_user(user_id)
);

/* ===============================================================
   TABEL: tm_livestock - Master data ternak (jika diperlukan per individu)
   =============================================================== */
create table tm_livestock
(
   animal_id            varchar(32) not null,
   tenant_id            varchar(32),
   site_id              varchar(32),
   cluster_id           varchar(32),
   animal_tag           varchar(64),
   animal_species       varchar(32),
   animal_breed         varchar(64),
   animal_gender        varchar(8),
   animal_birthdate     date,
   animal_status        varchar(16),  -- aktif / dijual / sakit
   primary key (animal_id),
   foreign key (tenant_id) references tm_tenant(tenant_id),
   foreign key (site_id) references tm_site(site_id),
   foreign key (cluster_id) references tm_cluster(cluster_id)
);

/* ===============================================================
   TABEL: tm_livestock_activity - Aktivitas harian ternak (pakan, vaksin, mortality)
   =============================================================== */
create table tm_livestock_activity
(
   activity_id          varchar(64) not null,
   tenant_id            varchar(32),
   animal_id            varchar(32),
   cluster_id           varchar(32),
   user_id              varchar(32),
   activity_type        varchar(32),  -- feeding / medical / movement / mortality
   activity_desc        varchar(255),
   activity_value       varchar(64),
   activity_date        datetime,
   primary key (activity_id),
   foreign key (tenant_id) references tm_tenant(tenant_id),
   foreign key (animal_id) references tm_livestock(animal_id),
   foreign key (cluster_id) references tm_cluster(cluster_id),
   foreign key (user_id) references tm_user(user_id)
);

/* ===============================================================
   INDEX-INDEX PENTING UNTUK PERFORMA
   =============================================================== */
create index idx_user_tenant on tm_user(tenant_id);
create index idx_site_tenant on tm_site(tenant_id);
create index idx_cluster_site on tm_cluster(site_id);
create index idx_device_tenant on tm_device(tenant_id);
create index idx_device_cluster on tm_device(cluster_id);
create index idx_td_ds_dev on td_device_sensor(dev_id);
create index idx_sensor_read_date on tm_sensor_read(read_date);
create index idx_sensor_read_ds on tm_sensor_read(ds_id);
create index idx_livestock_cluster on tm_livestock(cluster_id);
create index idx_activity_date on tm_livestock_activity(activity_date);
